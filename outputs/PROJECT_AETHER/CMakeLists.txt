cmake_minimum_required(VERSION 3.20)
project(ProjectAether VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB AETHER_CORE CONFIGURE_DEPENDS core/src/*.cpp)
add_library(aether_core ${AETHER_CORE})
target_include_directories(aether_core PUBLIC core/include)
if(MSVC)
  target_compile_options(aether_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(aether_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aether_cli application/main.cpp)
target_link_libraries(aether_cli PRIVATE aether_core)
file(GLOB AETHER_TESTS CONFIGURE_DEPENDS tests/*.cpp)
add_executable(aether_tests ${AETHER_TESTS})
target_link_libraries(aether_tests PRIVATE aether_core)
enable_testing()
add_test(NAME core COMMAND aether_tests)
set_tests_properties(core PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
